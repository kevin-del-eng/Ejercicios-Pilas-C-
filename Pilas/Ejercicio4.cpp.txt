#include <iostream>
#include <stack>
#include <string>
using namespace std;

int main() {
    stack<string> historial;
    int opcion;
    string accion;

    do {
        cin >> opcion;
        cin.ignore();

        switch (opcion) {
            case 1:
                getline(cin, accion);
                historial.push(accion);
                break;
            case 2:
                if (!historial.empty()) {
                    cout << historial.top() << endl;
                }
                break;
            case 3:
                if (!historial.empty()) {
                    historial.pop();
                }
                break;
        }
    } while (opcion != 4);

    return 0;
}
