#include <iostream>
#include <stack>
#include <string>
using namespace std;

bool estanBalanceados(string expr) {
    stack<char> pila;
    for (char c : expr) {
        if (c == '(') {
            pila.push(c);
        } else if (c == ')') {
            if (pila.empty()) {
                return false;
            }
            pila.pop();
        }
    }
    return pila.empty();
}

int main() {
    string expresion;
    cin >> expresion;

    if (estanBalanceados(expresion)) {
        cout << "Correcto" << endl;
    } else {
        cout << "Incorrecto" << endl;
    }

    return 0;
}
